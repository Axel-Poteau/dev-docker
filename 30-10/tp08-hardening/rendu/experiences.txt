### 6. --user 1000
$ docker run --rm --user 1000 alpine:3.20 id
  uid=1000 gid=0(root) groups=0(root)
  (rc=0)
$ docker run --rm -v exp-data:/data alpine:3.20 sh -c 'echo root > /data/f && ls -ln /data'
  total 4
  -rw-r--r--    1 0        0                5 Oct 30 08:43 f
  (rc=0)
$ docker run --rm --user 1000 -v exp-data:/data alpine:3.20 sh -c 'echo app >> /data/f'
  sh: can't create /data/f: Permission denied
  (rc=1)
$ docker run --rm -v exp-data:/data alpine:3.20 chown -R 1000:1000 /data
  (rc=0)
$ docker run --rm --user 1000 -v exp-data:/data alpine:3.20 sh -c 'echo app >> /data/f && cat /data/f'
  root
  app
  (rc=0)
